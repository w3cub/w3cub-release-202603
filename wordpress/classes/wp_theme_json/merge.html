/**
 * A public helper to get the block nodes from a theme.json file.
 *
 * @since 6.1.0
 *
 * @return array The block nodes in theme.json.
 */
public function get_styles_block_nodes() {
	return static::get_block_nodes( $this->theme_json );
}
